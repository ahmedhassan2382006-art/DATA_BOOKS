الحمد لله وحده وأشهد أن لا إله إلا الله وحده لا شريك له وأشهد أن محمدا عبد الله ورسوله صلى الله عليه وعلى آله وأصحابه ومن تبعه إلى يوم الدين وسلم تسليما كثيرا ... أما بعد
فهذا متن في الحديث مما يتعلق بالصيام وصدقة الفطر والعيد والإعتكاف بحيث أنني اخترت في ذلك كثيرا من الأحاديث الصحيحة أو الحسنة مما صححه أو حسنه بعض أهل العلم المعتبرين وهي أحاديث (بدون مكرر) وتصلح أن تكون متنا يحفظه الطلاب وغيرهم ويقوم بشرحه طلاب العلم واسأل الله أن يجعل هذا العمل خالصا لوجهه الكريم وأن ينفع به وأن يجعلنا من أتباع رسول الله صلى الله عليه وسلم في حياتنا كلها وأن يحسن لنا الخاتمة والله الموفق

فضل الصوم
- عَنْ سَهْلٍ رَضِيَ اللَّهُ عَنْهُ عَنْ النَّبِيِّ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ قَالَ ((إِنَّ فِي الْجَنَّةِ بَابًا يُقَالُ لَهُ الرَّيَّانُ يَدْخُلُ مِنْهُ الصَّائِمُونَ يَوْمَ الْقِيَامَةِ لَا يَدْخُلُ مِنْهُ أَحَدٌ غَيْرُهُمْ يُقَالُ أَيْنَ الصَّائِمُونَ فَيَقُومُونَ لَا يَدْخُلُ مِنْهُ أَحَدٌ غَيْرُهُمْ فَإِذَا دَخَلُوا أُغْلِقَ فَلَمْ يَدْخُلْ مِنْهُ أَحَدٌ)) رواه البخاري0
(1/1)

- عَنْ أَبِي هُرَيْرَةَ رَضِيَ اللَّهُ عَنْهُ قَالَ قَالَ رَسُولُ اللَّهِ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ ((كُلُّ عَمَلِ ابْنِ آدَمَ يُضَاعَفُ الْحَسَنَةُ عَشْرُ أَمْثَالِهَا إِلَى سَبْعمِائَة ضِعْفٍ قَالَ اللَّهُ عَزَّ وَجَلَّ إِلَّا الصَّوْمَ فَإِنَّهُ لِي وَأَنَا أَجْزِي بِهِ يَدَعُ شَهْوَتَهُ وَطَعَامَهُ مِنْ أَجْلِي لِلصَّائِمِ فَرْحَتَانِ فَرْحَةٌ عِنْدَ فِطْرِهِ وَفَرْحَةٌ عِنْدَ لِقَاءِ رَبِّهِ وَلَخُلُوفُ فِيهِ أَطْيَبُ عِنْدَ اللَّهِ مِنْ رِيحِ الْمِسْكِ)) رواه الشيخان وهذا لفظ مسلم-وعن أَبَي هُرَيْرَةَ عَنْ النَّبِيِّ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ يَرْوِيهِ عَنْ رَبِّكُمْ قَالَ ((لِكُلِّ عَمَلٍ كَفَّارَةٌ وَالصَّوْمُ لِي وَأَنَا أَجْزِي بِهِ وَلَخُلُوفُ فَمِ الصَّائِمِ أَطْيَبُ عِنْدَ اللَّهِ مِنْ رِيحِ الْمِسْكِ)) رواه البخاري0
-وعن أَبِي هُرَيْرَةَ رَضِيَ اللَّهُ عَنْهُ قَالَ رَسُولُ اللَّهِ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ قَالَ اللَّهُ ((كُلُّ عَمَلِ ابْنِ آدَمَ لَهُ إِلَّا الصِّيَامَ فَإِنَّهُ لِي وَأَنَا أَجْزِي بِهِ وَالصِّيَامُ جُنَّةٌ وَإِذَا كَانَ يَوْمُ صَوْمِ أَحَدِكُمْ فَلَا يَرْفُثْ وَلَا يَصْخَبْ فَإِنْ سَابَّهُ أَحَدٌ أَوْ قَاتَلَهُ فَلْيَقُلْ إِنِّي امْرُؤٌ صَائِمٌ وَالَّذِي نَفْسُ مُحَمَّدٍ بِيَدِهِ لَخُلُوفُ فَمِ الصَّائِمِ أَطْيَبُ عِنْدَ اللَّهِ مِنْ رِيحِ الْمِسْكِ لِلصَّائِمِ فَرْحَتَانِ يَفْرَحُهُمَا إِذَا أَفْطَرَ فَرِحَ وَإِذَا لَقِيَ رَبَّهُ فَرِحَ بِصَوْمِهِ)) رواه الشيخان0
(1/2)

- عَن مُطَرِّفٍ قَالَ دَخَلْتُ عَلَى عُثْمَانَ بْنِ أَبِي الْعَاصِ فَأَمَرَ لِي بِلَبَنِ لِقْحَةٍ فَقُلْتُ إِنِّي صَائِمٌ فَقَالَ سَمِعْتُ رَسُولَ اللَّهِ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ يَقُولُ ((الصَّوْمُ جُنَّةٌ مِنْ عَذَابِ اللَّهِ كَجُنَّةِ أَحَدِكُمْ مِنْ الْقِتَالِ وَصِيَامٌ حَسَنٌ ثَلَاثَةُ أَيَّامٍ مِنْ كُلِّ شَهْرٍ قَالَ وَكَانَ آخِرُ شَيْءٍ عَهِدَهُ النَّبِيُّ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ إِلَيَّ أَنْ قَالَ جَوِّزْ فِي صَلَاتِكَ وَاقْدُرْ النَّاسَ بِأَضْعَفِهِمْ فَإِنَّ مِنْهُمْ الصَّغِيرَ وَالْكَبِيرَ وَالضَّعِيفَ وَذَا الْحَاجَةِ)) رواه أحمد0
- وعن عَبْدِ اللَّهِ قال قَالَ لَنَا النَّبِيُّ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ ((يَا مَعْشَرَ الشَّبَابِ مَنْ اسْتَطَاعَ مِنْكُمْ الْبَاءَةَ فَلْيَتَزَوَّجْ فَإِنَّهُ أَغَضُّ لِلْبَصَرِ وَأَحْصَنُ لِلْفَرْجِ وَمَنْ لَمْ يَسْتَطِعْ فَعَلَيْهِ بِالصَّوْمِ فَإِنَّهُ لَهُ وِجَاءٌ)) الشيخان وهذا لفظ مسلم0
- عَنْ عَبْدِ اللَّهِ بْنِ عَمْرٍو أَنَّ رَسُولَ اللَّهِ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ قَالَ ((الصِّيَامُ وَالْقُرْآنُ يَشْفَعَانِ لِلْعَبْدِ يَوْمَ الْقِيَامَةِ يَقُولُ الصِّيَامُ أَيْ رَبِّ مَنَعْتُهُ الطَّعَامَ وَالشَّهَوَاتِ بِالنَّهَارِ فَشَفِّعْنِي فِيهِ وَيَقُولُ الْقُرْآنُ مَنَعْتُهُ النَّوْمَ بِاللَّيْلِ فَشَفِّعْنِي فِيهِ قَالَ فَيُشَفَّعَانِ)) رواه أحمد والبيهقي في الشعب والحاكم وقال هذا حديث صحيح على شرط مسلم ولم يخرجاه –حديث صحيح-0
(1/3)

- عَنْ عُبَيْدِ بْنِ خَالِدٍ السُّلَمِيِّ قَالَ ((آخَى رَسُولُ اللَّهِ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ بَيْنَ رَجُلَيْنِ فَقُتِلَ أَحَدُهُمَا وَمَاتَ الْآخَرُ بَعْدَهُ بِجُمُعَةٍ أَوْ نَحْوِهَا فَصَلَّيْنَا عَلَيْهِ فَقَالَ رَسُولُ اللَّهِ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ مَا قُلْتُمْ فَقُلْنَا دَعَوْنَا لَهُ وَقُلْنَا اللَّهُمَّ اغْفِرْ لَهُ وَأَلْحِقْهُ بِصَاحِبِهِ فَقَالَ رَسُولُ اللَّهِ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ فَأَيْنَ صَلَاتُهُ بَعْدَ صَلَاتِهِ وَصَوْمُهُ بَعْدَ صَوْمِهِ شَكَّ شُعْبَةُ فِي صَوْمِهِ وَعَمَلُهُ بَعْدَ عَمَلِهِ إِنَّ بَيْنَهُمَا كَمَا بَيْنَ السَّمَاءِ وَالْأَرْضِ)) رواه أبو داود وعند أحمد ((فَقُتِلَ أَحَدُهُمَا وَمَاتَ الْآخَرُ بَعْدَهُ فَصَلَّيْنَا عَلَيْهِ فَقَالَ رَسُولُ اللَّهِ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ مَا قُلْتُمْ قَالُوا دَعَوْنَا لَهُ اللَّهُمَّ أَلْحِقْهُ بِصَاحِبِهِ فَقَالَ رَسُولُ اللَّهِ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ فَأَيْنَ صَلَاتُهُ بَعْدَ صَلَاتِهِ وَأَيْنَ صَوْمُهُ بَعْدَ صَوْمِهِ وَأَيْنَ عَمَلُهُ بَعْدَ عَمَلِهِ شَكَّ فِي الصَّلَاةِ وَالْعَمَلِ شُعْبَةُ فِي أَحَدِهِمَا الَّذِي بَيْنَهُمَا كَمَا بَيْنَ السَّمَاءِ وَالْأَرْضِ)) 0
(1/4)

- عَنْ عَلِيٍّ - رضي الله عنه - قَالَ قَالَ النَّبِيُّ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ ((إِنَّ فِي الْجَنَّةِ غُرَفًا تُرَى ظُهُورُهَا مِنْ بُطُونِهَا وَبُطُونُهَا مِنْ ظُهُورِهَا فَقَامَ أَعْرَابِيٌّ فَقَالَ لِمَنْ هِيَ يَا رَسُولَ اللَّهِ قَالَ لِمَنْ أَطَابَ الْكَلَامَ وَأَطْعَمَ الطَّعَامَ وَأَدَامَ الصِّيَامَ وَصَلَّى لِلَّهِ بِاللَّيْلِ وَالنَّاسُ نِيَامٌ)) رواه الترمذي-حديث حسن-ورواه الطبراني في الكبير عن أَبُي مَالِكٍ الأَشْعَرِيُّ عن رَسُولَ اللَّهِ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ بنحوه ورواه الحاكم عَنْ عَبْدِ اللَّهِ بْنِ عَمْرٍو عن رَسُولَ اللَّهِ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ بنحوه وقال الحاكم هذا حديث صحيح على شرط الشيخين فقد احتجا جميعا بيحي وهو أبو عبد الرحمن المذحجي صاحب سليمان بن عبد الملك ويقال مولاه ولم يخرجاه0
- عَنْ أَبِي سَعِيدٍ الْخُدْرِيِّ رَضِيَ اللَّهُ عَنْهُ قَالَ سَمِعْتُ النَّبِيَّ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ يَقُولُ ((مَنْ صَامَ يَوْمًا فِي سَبِيلِ اللَّهِ بَعَّدَ اللَّهُ وَجْهَهُ عَنْ النَّارِ سَبْعِينَ خَرِيفًا)) رواه البخاري0

- وعَنْ أَسْمَاءَ بِنْتِ أَبِي بَكْرٍ الصِّدِّيقِ رَضِيَ اللَّهُ عَنْهُمَا قَالَتْ ((أَفْطَرْنَا عَلَى عَهْدِ النَّبِيِّ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ يَوْمَ غَيْمٍ ثُمَّ طَلَعَتْ الشَّمْسُ قِيلَ لِهِشَامٍ فَأُمِرُوا بِالْقَضَاءِ قَالَ لَا بُدَّ مِنْ قَضَاءٍ)) رواه البخاري0
(1/24)

- وعن عَائِشَةَ رَضِيَ اللَّهُ عَنْهَا قالت ((كَانَ يَكُونُ عَلَيَّ الصَّوْمُ مِنْ رَمَضَانَ فَمَا أَسْتَطِيعُ أَنْ أَقْضِيَ إِلَّا فِي شَعْبَانَ قَالَ يَحْيَى الشُّغْلُ مِنْ النَّبِيِّ أَوْ بِالنَّبِيِّ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ)) رواه البخاري0

- وعَنْ ابْنِ عَبَّاسٍ رَضِيَ اللَّهُ عَنْهُمَا ((أَنَّ امْرَأَةً أَتَتْ رَسُولَ اللَّهِ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ فَقَالَتْ إِنَّ أُمِّي مَاتَتْ وَعَلَيْهَا صَوْمُ شَهْرٍ فَقَالَ أَرَأَيْتِ لَوْ كَانَ عَلَيْهَا دَيْنٌ أَكُنْتِ تَقْضِينَهُ قَالَتْ نَعَمْ قَالَ فَدَيْنُ اللَّهِ أَحَقُّ بِالْقَضَاءِ)) رواه مسلم0
-وقال البخاري وَيُذْكَرُ عَنْ أَبِي هُرَيْرَةَ رَفَعَهُ ((مَنْ أَفْطَرَ يَوْمًا مِنْ رَمَضَانَ مِنْ غَيْرِ عُذْرٍ وَلَا مَرَضٍ لَمْ يَقْضِهِ صِيَامُ الدَّهْرِ وَإِنْ صَامَهُ))

-عن عَبْدَ اللَّهِ بْنَ عَمْرِو بْنِ الْعَاصِ رَضِيَ اللَّهُ عَنْهُمَا أَنَّ رَسُولَ اللَّهِ صَلَّى اللَّهُ عَلَيْهِ وَسَلَّمَ قَالَ لَهُ ((أَحَبُّ الصَّلَاةِ إِلَى اللَّهِ صَلَاةُ دَاوُدَ عَلَيْهِ السَّلَام وَأَحَبُّ الصِّيَامِ إِلَى اللَّهِ صِيَامُ دَاوُدَ وَكَانَ يَنَامُ نِصْفَ اللَّيْلِ وَيَقُومُ ثُلُثَهُ وَيَنَامُ سُدُسَهُ وَيَصُومُ يَوْمًا وَيُفْطِرُ يَوْمًا)) رواه الشيخان0
